% Неформальная постановка и актуальность
\begin{frame}{Задача и её актуальность}\relax
\textbf{Задача:} по сетке STL восстановить редактируемую CAD-программу.
\par\bigskip
\textbf{Актуальность:} меньше ручной работы при изменении готовой формы.
\par\bigskip
\textbf{Проверяем:} качество и цену недорогой модели с обратной связью.
\par\medskip
Два домена: механические детали и предметы интерьера.
\source{Исходную историю построения восстанавливать не требуется.}
\end{frame}

\begin{frame}{Формальная постановка задачи}\relax
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
  \textbf{Вход: треугольная сетка}
  \[
  M=(V,F),\quad V\in\mathbb R^{n\times3},
  \]
  \[
  F\in\{1,\ldots,n\}^{m\times3}.
  \]
  {\smalltext Замкнутая ориентированная сетка задаёт
  объём $\Omega_M\subset\mathbb R^3$, $\operatorname{Vol}(\Omega_M)>0$.}
\end{column}
\begin{column}{.48\textwidth}
  \textbf{Выход: исполнимая программа}
  \[
  p\in\mathcal P_{\mathrm{CadQuery}},\quad E(p)=B_p.
  \]
  {\smalltext $B_p$ задаёт объём $\Omega_p$.
  Требования: валидное твёрдое тело, положительный объём,
  координаты и масштаб входа.}
\end{column}
\end{columns}
\par\medskip
\[
\max_{p:\,E(p)\ \text{валиден}} J(M,p),\qquad
J(M,p)=\frac{\operatorname{Vol}(\Omega_M\cap\Omega_p)}
{\operatorname{Vol}(\Omega_M\cup\Omega_p)}.
\]
{\smalltext Лимиты: $K\leq5$ генераций и бюджет вычислений.
Цель $J\geq0.9$ не гарантируется.}
\source{Выход поиска: лучшая проверенная программа или статус ошибки.}
\end{frame}

\begin{frame}{Проблема и наш подход}\relax
\textbf{Проблема:} исполнимый код может давать неверную геометрию.
\par\medskip
\textbf{Решение:} модель исправляет код по IoU, карте расхождений
и ошибкам исполнения.
\par\medskip
Готовая модель, без обучения весов. До пяти генераций.
\par\medskip
\alert{Гипотеза: качество близко к дорогой модели, цена API ниже.}
\source{Обратную связь уже используют CADENA и CADFather.
Наш вклад: ограниченный цикл и парная оценка в двух доменах.}
\end{frame}

% 10. Метрики
\begin{frame}{Метрики и правила подсчёта}\relax
\begin{columns}[T,onlytextwidth]
\begin{column}{.52\textwidth}
  \textbf{Качество формы}
  \[
  \IoU(M,\widehat M)=
  \frac{\operatorname{Vol}(M\cap\widehat M)}{\operatorname{Vol}(M\cup\widehat M)}.
  \]
  {\smalltext
  Полные объёмы, без выравнивания.
  \par\medskip
  Неудачная реконструкция: IoU = 0.}
\end{column}
\begin{column}{.43\textwidth}
  \textbf{Дополнительно}
  \begin{itemize}
    \item Валидность CAD.
    \item Доля IoU $\geq0.9$.
    \item Цена API, вызовы и время.
  \end{itemize}
  \par\smallskip
  {\smalltext Harness: лучший IoU, цена всех попыток.}
\end{column}
\end{columns}
\source{Непригодные входы отобраны до запуска. Не начатые объекты не включены в среднее.}
\end{frame}
